generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model ConfigSistema {
  id                  Int     @id @default(1)
  modoAtivo           String  @default("BARBEARIA")
  nomeAssistente      String  @default("Assistente")
  tomDeVoz            String  @default("Profissional")
  objetivos           String? @db.Text
  regrasExtrasIA      String? @db.Text
  faq                 String? @db.Text
  metaToken           String? @db.Text
  webhookUrl          String?
  notificarNovosLeads Boolean @default(true)
}

model Usuario {
  id          Int           @id @default(autoincrement())
  nome        String
  email       String        @unique
  senha       String
  funcao      String        @default("ATENDENTE") 
  status      String        @default("ONLINE")
  notas       NotaInterna[]
  criadoEm    DateTime      @default(now())
}

model Cliente {
  id              String        @id // Número de telefone via Meta API
  nome            String?
  email           String?
  fotoUrl         String?       @db.Text 
  leadStatus      String        @default("NOVO") 
  falarHumano     Boolean       @default(false)
  tags            String?
  valorPotencial  Float         @default(0.0)
  origem          String        @default("WhatsApp Meta")
  observacoes     String?       @db.Text
  ultimaInteracao DateTime      @default(now())
  criadoEm        DateTime      @default(now())
  mensagens       MensagemIA[]
  agendamentos    Agendamento[]
  notas           NotaInterna[]
}

model MensagemIA {
  id              Int      @id @default(autoincrement())
  clienteId       String
  cliente         Cliente  @relation(fields: [clienteId], references: [id], onDelete: Cascade)
  role            String   
  content         String   @db.Text
  midiaUrl        String?  @db.Text 
  tipoMidia       String?  
  atendenteHumano Boolean  @default(false) 
  criadoEm        DateTime @default(now())
}

model Tratamento {
  id           Int           @id @default(autoincrement())
  nome         String
  descricao    String?       @db.Text
  preco        Float
  duracaoMin   Int           @default(30)
  imagemUrl    String?       @db.Text 
  faq          String?       @db.Text
  status       String        @default("ATIVO")
  criadoEm     DateTime      @default(now())
  agendamentos Agendamento[]
}

model ProfissionalSaude {
  id            Int           @id @default(autoincrement())
  nome          String
  especialidade String
  agendamentos  Agendamento[]
}

model Servico {
  id           Int           @id @default(autoincrement())
  nome         String
  preco        Float
  duracaoMin   Int           @default(30)
  agendamentos Agendamento[]
}

model Barbeiro {
  id           Int           @id @default(autoincrement())
  nome         String
  agendamentos Agendamento[]
}

model Agendamento {
  id                  Int                @id @default(autoincrement())
  dataHora            DateTime
  status              String             @default("AGENDADO") 
  clienteId           String
  cliente             Cliente            @relation(fields: [clienteId], references: [id])
  
  tratamentoId        Int?
  tratamento          Tratamento?        @relation(fields: [tratamentoId], references: [id])
  profissionalSaudeId Int?
  profissionalSaude   ProfissionalSaude? @relation(fields: [profissionalSaudeId], references: [id])
  
  servicoId           Int?
  servico             Servico?           @relation(fields: [servicoId], references: [id])
  barbeiroId          Int?
  barbeiro            Barbeiro?          @relation(fields: [barbeiroId], references: [id])
  
  criadoEm            DateTime           @default(now())
  atualizadoEm        DateTime           @updatedAt
}

model NotaInterna {
  id        Int      @id @default(autoincrement())
  texto     String   @db.Text
  clienteId String
  cliente   Cliente  @relation(fields: [clienteId], references: [id], onDelete: Cascade)
  usuarioId Int
  usuario   Usuario  @relation(fields: [usuarioId], references: [id])
  criadoEm  DateTime @default(now())
}

model WebhookEndpoint {
  id        Int      @id @default(autoincrement())
  url       String   @db.Text
  secret    String?
  eventos   String   
  ativo     Boolean  @default(true)
  criadoEm  DateTime @default(now())
}

// === TABELA DE AUTOMAÇÕES ===
model Automacao {
  id        Int      @id @default(autoincrement())
  gatilho   String
  acao      String
  parametro String?  @db.Text
  ativo     Boolean  @default(true)
  criadoEm  DateTime @default(now())
}
